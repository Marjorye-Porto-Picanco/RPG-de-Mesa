-- ============================================================
-- CRIAÇÃO DO BANCO DE DADOS
-- ============================================================
IF NOT EXISTS (SELECT * FROM sys.databases WHERE name = 'rpg_system')
BEGIN
    CREATE DATABASE rpg_system;
END;
GO

USE rpg_system;
GO

-- ============================================================
-- 1. CRIAÇÃO DE TODAS AS TABELAS (SEM FOREIGN KEYS)
-- ============================================================

-- 1.1 Usuario
CREATE TABLE Usuario (
    id_account INT IDENTITY(1,1) NOT NULL,
    usar_name VARCHAR(50) NOT NULL UNIQUE,
    e_mail VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    data_cadastro DATETIME DEFAULT GETDATE(),
    status_conta VARCHAR(20) DEFAULT 'ativo',
    CONSTRAINT PK_Usuario PRIMARY KEY (id_account)
);

-- 1.2 Perfil
CREATE TABLE Perfil (
    id_perfil INT IDENTITY(1,1) NOT NULL,
    id_account INT NOT NULL UNIQUE,
    biografia VARCHAR(MAX),
    avatar_url VARCHAR(255),
    data_criacao DATETIME DEFAULT GETDATE(),
    CONSTRAINT PK_Perfil PRIMARY KEY (id_perfil)
);

-- 1.3 Raca
CREATE TABLE Raca (
    id_raca INT IDENTITY(1,1) NOT NULL,
    nome VARCHAR(50) NOT NULL,
    tamanho VARCHAR(20),
    deslocamento_base INT NOT NULL,
    visao_no_escuro BIT DEFAULT 0,
    CONSTRAINT PK_Raca PRIMARY KEY (id_raca)
);

-- 1.4 Antecedentes
CREATE TABLE Antecedentes (
    id_antecedente INT IDENTITY(1,1) NOT NULL,
    nome VARCHAR(50) NOT NULL,
    proficiencias_pericias VARCHAR(MAX),
    equipamento_inicial VARCHAR(MAX),
    CONSTRAINT PK_Antecedentes PRIMARY KEY (id_antecedente)
);

-- 1.5 Regras
CREATE TABLE Regras (
    id_regra INT IDENTITY(1,1) NOT NULL,
    livro_origem VARCHAR(100),
    capitulo VARCHAR(50),
    conteudo_texto VARCHAR(MAX),
    CONSTRAINT PK_Regras PRIMARY KEY (id_regra)
);

-- 1.6 Item
CREATE TABLE Item (
    id_item INT IDENTITY(1,1) NOT NULL,
    nome VARCHAR(100) NOT NULL,
    tipo VARCHAR(50),
    peso DECIMAL(5,2),
    preco DECIMAL(10,2),
    dano VARCHAR(20),
    tipo_dano VARCHAR(30),
    propriedades VARCHAR(MAX),
    ca_bonus INT DEFAULT 0,
    descricao VARCHAR(MAX),
    CONSTRAINT PK_Item PRIMARY KEY (id_item)
);

-- 1.7 Magia
CREATE TABLE Magia (
    id_magia INT IDENTITY(1,1) NOT NULL,
    nome VARCHAR(100) NOT NULL,
    circulo_nivel INT NOT NULL,
    escola VARCHAR(50),
    tempo_conjuracao VARCHAR(50),
    alcance VARCHAR(50),
    componentes VARCHAR(100),
    duracao VARCHAR(50),
    CONSTRAINT PK_Magia PRIMARY KEY (id_magia)
);

-- 1.8 Atributos
CREATE TABLE Atributos (
    id_atributo INT IDENTITY(1,1) NOT NULL,
    nome_atributo VARCHAR(50) NOT NULL,
    sigla VARCHAR(5) NOT NULL,
    valor_bruto INT NOT NULL,
    modificador INT NOT NULL,
    CONSTRAINT PK_Atributos PRIMARY KEY (id_atributo)
);

-- 1.9 Habilidades
CREATE TABLE Habilidades (
    id_habilidades INT IDENTITY(1,1) NOT NULL,
    nome VARCHAR(100) NOT NULL,
    tipo VARCHAR(50),
    descricao VARCHAR(MAX),
    custo_recurso VARCHAR(50),
    CONSTRAINT PK_Habilidades PRIMARY KEY (id_habilidades)
);

-- 1.10 Condicao
CREATE TABLE Condicao (
    id_condicao INT IDENTITY(1,1) NOT NULL,
    nome_condicao VARCHAR(50) NOT NULL,
    efeito_condicao VARCHAR(MAX),
    duracao_turnos INT,
    teste_resistencia VARCHAR(50),
    CONSTRAINT PK_Condicao PRIMARY KEY (id_condicao)
);

-- 1.11 Monstro
CREATE TABLE Monstro (
    id_monstro INT IDENTITY(1,1) NOT NULL,
    id_account INT NULL,
    nome VARCHAR(100) NOT NULL,
    tipo_monstro VARCHAR(50),
    nivel_desafio DECIMAL(4,2),
    pv_maximo INT NOT NULL,
    classe_armadura INT NOT NULL,
    tamanho VARCHAR(20),
    tendencia VARCHAR(40) NULL,
    deslocamento VARCHAR(50) NULL,
    sentidos VARCHAR(80) NULL,
    resistencias VARCHAR(120) NULL,
    habilidades VARCHAR(160) NULL,
    descricao VARCHAR(1000) NULL,
    imagem VARCHAR(255) NULL,
    CONSTRAINT PK_Monstro PRIMARY KEY (id_monstro)
);

-- 1.12 Acao_Combate
CREATE TABLE Acao_Combate (
    id_acao INT IDENTITY(1,1) NOT NULL,
    tipo_acao VARCHAR(50),
    nome_acao VARCHAR(100) NOT NULL,
    alcance_metros DECIMAL(5,2),
    descricao VARCHAR(MAX),
    CONSTRAINT PK_Acao_Combate PRIMARY KEY (id_acao)
);

-- 1.13 Campanha
CREATE TABLE Campanha (
    id_campanha INT IDENTITY(1,1) NOT NULL,
    id_account INT NOT NULL,
    nome VARCHAR(100) NOT NULL,
    data_criacao DATETIME DEFAULT GETDATE(),
    descricao VARCHAR(MAX),
    codigo_convite VARCHAR(20) NULL,
    CONSTRAINT PK_Campanha PRIMARY KEY (id_campanha)
);

-- 1.14 Mapa
CREATE TABLE Mapa (
    id_mapa INT IDENTITY(1,1) NOT NULL,
    id_campanha INT NOT NULL,
    nome_regiao VARCHAR(100) NOT NULL,
    escala VARCHAR(20),
    imagem_url VARCHAR(255),
    CONSTRAINT PK_Mapa PRIMARY KEY (id_mapa)
);

-- 1.15 Sessao
CREATE TABLE Sessao (
    id_sessao INT IDENTITY(1,1) NOT NULL,
    id_campanha INT NOT NULL,
    titulo_sessao VARCHAR(100) NOT NULL,
    data DATE NOT NULL,
    hora_inicio TIME,
    hora_fim TIME,
    resumo VARCHAR(MAX),
    CONSTRAINT PK_Sessao PRIMARY KEY (id_sessao)
);

-- 1.16 NPC
CREATE TABLE NPC (
    id_npc INT IDENTITY(1,1) NOT NULL,
    id_campanha INT NOT NULL,
    nome VARCHAR(100) NOT NULL,
    ocupacao VARCHAR(50),
    alinhamento VARCHAR(50),
    localizacao VARCHAR(100),
    descricao VARCHAR(MAX),
    CONSTRAINT PK_NPC PRIMARY KEY (id_npc)
);

-- 1.17 Classe
CREATE TABLE Classe (
    id_classe INT IDENTITY(1,1) NOT NULL,
    nome VARCHAR(50) NOT NULL,
    dado_vida VARCHAR(10) NOT NULL,
    atributo_primario VARCHAR(50),
    CONSTRAINT PK_Classe PRIMARY KEY (id_classe)
);

-- 1.18 Especialidade
CREATE TABLE Especialidade (
    id_especialidade INT IDENTITY(1,1) NOT NULL,
    id_classe INT NOT NULL,
    nome_especialidade VARCHAR(100) NOT NULL,
    nivel_desbloqueio INT NOT NULL,
    bonus_passivo VARCHAR(MAX),
    descricao VARCHAR(MAX),
    CONSTRAINT PK_Especialidade PRIMARY KEY (id_especialidade)
);

-- 1.19 Personagem (Atualizado com colunas completas da ficha e sem id_campanha direto)
CREATE TABLE Personagem (
    id_personagem INT IDENTITY(1,1) NOT NULL,
    id_account INT NOT NULL,
    id_raca INT NULL,
    id_antecedente INT NULL,
    nome VARCHAR(100) NOT NULL,
    nivel INT DEFAULT 1,
    experiencia_xp INT DEFAULT 0,
    pv_atual INT NOT NULL,
    pv_maximo INT NOT NULL,
    iniciativa INT DEFAULT 0,
    deslocamento INT DEFAULT 9,
    raca_texto VARCHAR(50) NULL,
    classe VARCHAR(50) NULL,
    antecedente_texto VARCHAR(50) NULL,
    tendencia VARCHAR(50) NULL,
    historia VARCHAR(MAX) NULL,
    imagem VARCHAR(255) NULL DEFAULT 'img/armaduraprata.png',
    forca INT DEFAULT 10,
    destreza INT DEFAULT 10,
    constituicao INT DEFAULT 10,
    inteligencia INT DEFAULT 10,
    sabedoria INT DEFAULT 10,
    carisma INT DEFAULT 10,
    idade VARCHAR(50) NULL,
    genero VARCHAR(50) NULL,
    altura VARCHAR(50) NULL,
    peso VARCHAR(50) NULL,
    aparencia VARCHAR(MAX) NULL,
    aliados VARCHAR(MAX) NULL,
    caracteristicas VARCHAR(MAX) NULL,
    tesouro VARCHAR(MAX) NULL,
    CONSTRAINT PK_Personagem PRIMARY KEY (id_personagem)
);

-- 1.20 Inventario
CREATE TABLE Inventario (
    id_inventario INT IDENTITY(1,1) NOT NULL,
    id_personagem INT NOT NULL UNIQUE,
    capacidade_carga DECIMAL(6,2),
    peso_atual DECIMAL(6,2) DEFAULT 0.00,
    dinheiro DECIMAL(10,2) DEFAULT 0.00,
    CONSTRAINT PK_Inventario PRIMARY KEY (id_inventario)
);

-- 1.21 Equipamento
CREATE TABLE Equipamento (
    id_equipamento INT IDENTITY(1,1) NOT NULL,
    id_inventario INT NOT NULL,
    durabilidade_atual INT,
    slot_corpo VARCHAR(50),
    CONSTRAINT PK_Equipamento PRIMARY KEY (id_equipamento)
);

-- 1.22 Personagem_Campanha (Com chave composta permitindo id_personagem nulo opcional)
CREATE TABLE Personagem_Campanha (
    id_personagem INT NULL,
    id_campanha INT NOT NULL,
    CONSTRAINT PK_Personagem_Campanha PRIMARY KEY (id_campanha, id_personagem)
);

-- 1.23 Usuario_Campanha
CREATE TABLE Usuario_Campanha (
    id_account INT NOT NULL,
    id_campanha INT NOT NULL,
    CONSTRAINT PK_Usuario_Campanha PRIMARY KEY (id_account, id_campanha),
    CONSTRAINT FK_UC_Usuario FOREIGN KEY (id_account) REFERENCES Usuario(id_account) ON DELETE CASCADE,
    CONSTRAINT FK_UC_Campanha FOREIGN KEY (id_campanha) REFERENCES Campanha(id_campanha) ON DELETE CASCADE
);

-- 1.24 Personagem_Classe
CREATE TABLE Personagem_Classe (
    id_personagem INT NOT NULL,
    id_classe INT NOT NULL,
    nivel_na_classe INT DEFAULT 1,
    CONSTRAINT PK_Personagem_Classe PRIMARY KEY (id_personagem, id_classe)
);

-- 1.25 Personagem_Atributos
CREATE TABLE Personagem_Atributos (
    id_personagem INT NOT NULL,
    id_atributo INT NOT NULL,
    CONSTRAINT PK_Personagem_Atributos PRIMARY KEY (id_personagem, id_atributo)
);

-- 1.26 Personagem_Magia
CREATE TABLE Personagem_Magia (
    id_personagem INT NOT NULL,
    id_magia INT NOT NULL,
    CONSTRAINT PK_Personagem_Magia PRIMARY KEY (id_personagem, id_magia)
);

-- 1.27 Personagem_Habilidades
CREATE TABLE Personagem_Habilidades (
    id_personagem INT NOT NULL,
    id_habilidades INT NOT NULL,
    CONSTRAINT PK_Personagem_Habilidades PRIMARY KEY (id_personagem, id_habilidades)
);

-- 1.28 Personagem_Acao
CREATE TABLE Personagem_Acao (
    id_personagem INT NOT NULL,
    id_acao INT NOT NULL,
    CONSTRAINT PK_Personagem_Acao PRIMARY KEY (id_personagem, id_acao)
);

-- 1.29 Inventario_Item
CREATE TABLE Inventario_Item (
    id_inventario INT NOT NULL,
    id_item INT NOT NULL,
    quantidade INT DEFAULT 1,
    CONSTRAINT PK_Inventario_Item PRIMARY KEY (id_inventario, id_item)
);

-- 1.30 Equipamento_Item
CREATE TABLE Equipamento_Item (
    id_equipamento INT NOT NULL,
    id_item INT NOT NULL,
    CONSTRAINT PK_Equipamento_Item PRIMARY KEY (id_equipamento, id_item)
);

-- 1.31 Monstro_Campanha
CREATE TABLE Monstro_Campanha (
    id_monstro INT NOT NULL,
    id_campanha INT NOT NULL,
    CONSTRAINT PK_Monstro_Campanha PRIMARY KEY (id_monstro, id_campanha)
);

-- 1.32 Monstro_Regras
CREATE TABLE Monstro_Regras (
    id_monstro INT NOT NULL,
    id_regra INT NOT NULL,
    CONSTRAINT PK_Monstro_Regras PRIMARY KEY (id_monstro, id_regra)
);

-- 1.33 Monstro_Condicao
CREATE TABLE Monstro_Condicao (
    id_monstro INT NOT NULL,
    id_condicao INT NOT NULL,
    CONSTRAINT PK_Monstro_Condicao PRIMARY KEY (id_monstro, id_condicao)
);

-- 1.34 Personagem_Condicao
CREATE TABLE Personagem_Condicao (
    id_personagem INT NOT NULL,
    id_condicao INT NOT NULL,
    CONSTRAINT PK_Personagem_Condicao PRIMARY KEY (id_personagem, id_condicao)
);
GO

-- ============================================================
-- 2. ADIÇÃO DE FOREIGN KEYS COM ALTER TABLE ... ADD CONSTRAINT
-- ============================================================

ALTER TABLE Perfil ADD CONSTRAINT FK_Perfil_Usuario FOREIGN KEY (id_account) REFERENCES Usuario(id_account) ON DELETE CASCADE;
ALTER TABLE Monstro ADD CONSTRAINT FK_Monstro_Usuario FOREIGN KEY (id_account) REFERENCES Usuario(id_account);
ALTER TABLE Campanha ADD CONSTRAINT FK_Campanha_Usuario FOREIGN KEY (id_account) REFERENCES Usuario(id_account);
ALTER TABLE Mapa ADD CONSTRAINT FK_Mapa_Campanha FOREIGN KEY (id_campanha) REFERENCES Campanha(id_campanha) ON DELETE CASCADE;
ALTER TABLE Sessao ADD CONSTRAINT FK_Sessao_Campanha FOREIGN KEY (id_campanha) REFERENCES Campanha(id_campanha) ON DELETE CASCADE;
ALTER TABLE NPC ADD CONSTRAINT FK_NPC_Campanha FOREIGN KEY (id_campanha) REFERENCES Campanha(id_campanha) ON DELETE CASCADE;
ALTER TABLE Especialidade ADD CONSTRAINT FK_Especialidade_Classe FOREIGN KEY (id_classe) REFERENCES Classe(id_classe) ON DELETE CASCADE;

ALTER TABLE Personagem ADD CONSTRAINT FK_Personagem_Usuario FOREIGN KEY (id_account) REFERENCES Usuario(id_account),
                       CONSTRAINT FK_Personagem_Raca FOREIGN KEY (id_raca) REFERENCES Raca(id_raca),
                       CONSTRAINT FK_Personagem_Antecedentes FOREIGN KEY (id_antecedente) REFERENCES Antecedentes(id_antecedente);

ALTER TABLE Inventario ADD CONSTRAINT FK_Inventario_Personagem FOREIGN KEY (id_personagem) REFERENCES Personagem(id_personagem) ON DELETE CASCADE;
ALTER TABLE Equipamento ADD CONSTRAINT FK_Equipamento_Inventario FOREIGN KEY (id_inventario) REFERENCES Inventario(id_inventario) ON DELETE CASCADE;

ALTER TABLE Personagem_Campanha ADD CONSTRAINT FK_PC_Personagem FOREIGN KEY (id_personagem) REFERENCES Personagem(id_personagem) ON DELETE CASCADE,
                                CONSTRAINT FK_PC_Campanha FOREIGN KEY (id_campanha) REFERENCES Campanha(id_campanha) ON DELETE CASCADE;

ALTER TABLE Personagem_Classe ADD CONSTRAINT FK_PClass_Personagem FOREIGN KEY (id_personagem) REFERENCES Personagem(id_personagem) ON DELETE CASCADE,
                              CONSTRAINT FK_PClass_Classe FOREIGN KEY (id_classe) REFERENCES Classe(id_classe) ON DELETE CASCADE;

ALTER TABLE Personagem_Atributos ADD CONSTRAINT FK_PAttr_Personagem FOREIGN KEY (id_personagem) REFERENCES Personagem(id_personagem) ON DELETE CASCADE,
                                 CONSTRAINT FK_PAttr_Atributos FOREIGN KEY (id_atributo) REFERENCES Atributos(id_atributo) ON DELETE CASCADE;

ALTER TABLE Personagem_Magia ADD CONSTRAINT FK_PMag_Personagem FOREIGN KEY (id_personagem) REFERENCES Personagem(id_personagem) ON DELETE CASCADE,
                             CONSTRAINT FK_PMag_Magia FOREIGN KEY (id_magia) REFERENCES Magia(id_magia) ON DELETE CASCADE;

ALTER TABLE Personagem_Habilidades ADD CONSTRAINT FK_PHab_Personagem FOREIGN KEY (id_personagem) REFERENCES Personagem(id_personagem) ON DELETE CASCADE,
                                   CONSTRAINT FK_PHab_Habilidades FOREIGN KEY (id_habilidades) REFERENCES Habilidades(id_habilidades) ON DELETE CASCADE;

ALTER TABLE Personagem_Acao ADD CONSTRAINT FK_PAcao_Personagem FOREIGN KEY (id_personagem) REFERENCES Personagem(id_personagem) ON DELETE CASCADE,
                            CONSTRAINT FK_PAcao_Acao FOREIGN KEY (id_acao) REFERENCES Acao_Combate(id_acao) ON DELETE CASCADE;

ALTER TABLE Inventario_Item ADD CONSTRAINT FK_InvItem_Inventario FOREIGN KEY (id_inventario) REFERENCES Inventario(id_inventario) ON DELETE CASCADE,
                            CONSTRAINT FK_InvItem_Item FOREIGN KEY (id_item) REFERENCES Item(id_item) ON DELETE CASCADE;

ALTER TABLE Equipamento_Item ADD CONSTRAINT FK_EqpItem_Equipamento FOREIGN KEY (id_equipamento) REFERENCES Equipamento(id_equipamento) ON DELETE CASCADE,
                             CONSTRAINT FK_EqpItem_Item FOREIGN KEY (id_item) REFERENCES Item(id_item) ON DELETE CASCADE;

ALTER TABLE Monstro_Campanha ADD CONSTRAINT FK_MCamp_Monstro FOREIGN KEY (id_monstro) REFERENCES Monstro(id_monstro) ON DELETE CASCADE,
                             CONSTRAINT FK_MCamp_Campanha FOREIGN KEY (id_campanha) REFERENCES Campanha(id_campanha) ON DELETE CASCADE;

ALTER TABLE Monstro_Regras ADD CONSTRAINT FK_MReg_Monstro FOREIGN KEY (id_monstro) REFERENCES Monstro(id_monstro) ON DELETE CASCADE,
                           CONSTRAINT FK_MReg_Regras FOREIGN KEY (id_regra) REFERENCES Regras(id_regra) ON DELETE CASCADE;

ALTER TABLE Monstro_Condicao ADD CONSTRAINT FK_MCond_Monstro FOREIGN KEY (id_monstro) REFERENCES Monstro(id_monstro) ON DELETE CASCADE,
                             CONSTRAINT FK_MCond_Condicao FOREIGN KEY (id_condicao) REFERENCES Condicao(id_condicao) ON DELETE CASCADE;

ALTER TABLE Personagem_Condicao ADD CONSTRAINT FK_PCond_Personagem FOREIGN KEY (id_personagem) REFERENCES Personagem(id_personagem) ON DELETE CASCADE,
                                CONSTRAINT FK_PCond_Condicao FOREIGN KEY (id_condicao) REFERENCES Condicao(id_condicao) ON DELETE CASCADE;
GO
